package backend

import (
	"context"
	"errors"
	"io"
	"net/http"
	"os"
	"strconv"
	"sync"
	"sync/atomic"
	"syscall"
	"time"
)

type gunshotCountedBody struct {
	io.ReadCloser
	bytes  atomic.Int64
	upload bool
}

func (b *gunshotCountedBody) Read(p []byte) (int, error) {
	n, err := b.ReadCloser.Read(p)
	b.bytes.Add(int64(n))
	if b.upload && n > 0 {
		gunshotRecordUploadBytes(int64(n))
	}
	return n, err
}

type gunshotRequestMetric struct {
	Requests, Errors, Milliseconds, BodyBytes int64
	HTTP2                                     int64
	Outcomes                                  map[string]int64 // status code or error class
}

var gunshotMetrics struct {
	sync.Mutex
	kinds             map[string]gunshotRequestMetric
	authCalls, authMS int64
	started           time.Time
	recent            [60]struct{ second, bytes int64 }
}
var gunshotUploadBytes atomic.Int64
var gunshotActiveUploads atomic.Int64

// Count body bytes as the transport reads them, including requests still in flight.
// These are bytes handed to HTTP, not acknowledgement of durable server storage.
func gunshotRecordUploadBytes(n int64) {
	gunshotUploadBytes.Add(n)
	gunshotMetrics.Lock()
	defer gunshotMetrics.Unlock()
	now := time.Now()
	if gunshotMetrics.started.IsZero() {
		gunshotMetrics.started = now
	}
	second := now.Unix()
	slot := &gunshotMetrics.recent[second%60]
	if slot.second != second {
		slot.second = second
		slot.bytes = 0
	}
	slot.bytes += n
}

func gunshotRequestKind(r *http.Request) string {
	if r.Header.Get("X-Goog-Upload-Command") == "query" {
		return "uploadQuery"
	}
	if r.URL.Hostname() == "photos.googleapis.com" && r.URL.Path == "/data/upload/uploadmedia/interactive" {
		if r.URL.Query().Get("upload_id") != "" {
			return "uploadBody"
		}
		return "uploadSession"
	}
	if r.URL.Hostname() == "photosdata-pa.googleapis.com" {
		return "mediaRPC"
	}
	return "other"
}

// gunshotRequestOutcome names why a request ended without URLs or payloads.
func gunshotRequestOutcome(response *http.Response, err error) string {
	switch {
	case err == nil && response != nil:
		return strconv.Itoa(response.StatusCode)
	case errors.Is(err, context.Canceled):
		return "canceled"
	case errors.Is(err, context.DeadlineExceeded) || os.IsTimeout(err):
		return "timeout"
	case errors.Is(err, syscall.ECONNRESET):
		return "reset"
	case errors.Is(err, io.EOF) || errors.Is(err, io.ErrUnexpectedEOF):
		return "eof"
	default:
		return "error"
	}
}
func gunshotRecordRequest(kind string, started time.Time, n int64, response *http.Response, err error) {
	gunshotMetrics.Lock()
	defer gunshotMetrics.Unlock()
	if gunshotMetrics.kinds == nil {
		gunshotMetrics.kinds = map[string]gunshotRequestMetric{}
		gunshotMetrics.started = started
	}
	value := gunshotMetrics.kinds[kind]
	value.Requests++
	value.Milliseconds += time.Since(started).Milliseconds()
	value.BodyBytes += n
	if err != nil || response == nil || response.StatusCode >= 400 {
		value.Errors++
	}
	if value.Outcomes == nil {
		value.Outcomes = map[string]int64{}
	}
	value.Outcomes[gunshotRequestOutcome(response, err)]++
	if response != nil && response.ProtoMajor == 2 {
		value.HTTP2++
	}
	gunshotMetrics.kinds[kind] = value
}
func gunshotRecordAuthorization(started time.Time) {
	gunshotMetrics.Lock()
	gunshotMetrics.authCalls++
	gunshotMetrics.authMS += time.Since(started).Milliseconds()
	gunshotMetrics.Unlock()
}
func GunshotTransportSnapshot() map[string]any {
	gunshotMetrics.Lock()
	defer gunshotMetrics.Unlock()
	requests := map[string]any{}
	for kind, value := range gunshotMetrics.kinds {
		outcomes := map[string]int64{}
		for outcome, count := range value.Outcomes {
			outcomes[outcome] = count
		}
		requests[kind] = map[string]any{"requests": value.Requests, "errors": value.Errors, "totalMs": value.Milliseconds, "meanMs": value.Milliseconds / max(1, value.Requests), "requestBodyBytes": value.BodyBytes, "http2Requests": value.HTTP2, "outcomes": outcomes}
	}
	now := time.Now().Unix()
	var sent, shortSent int64
	for _, slot := range gunshotMetrics.recent {
		age := now - slot.second
		if age >= 0 && age < 60 {
			sent += slot.bytes
		}
		if age >= 0 && age < 5 {
			shortSent += slot.bytes
		}
	}
	seconds := 60.0
	if !gunshotMetrics.started.IsZero() {
		seconds = min(seconds, max(1, time.Since(gunshotMetrics.started).Seconds()))
	}
	return map[string]any{"requests": requests, "nativeAuthCalls": gunshotMetrics.authCalls, "nativeAuthMeanMs": gunshotMetrics.authMS / max(1, gunshotMetrics.authCalls), "recentUploadBodyBytesPerSecond": int64(float64(sent) / seconds), "shortUploadBodyBytesPerSecond": int64(float64(shortSent) / min(5, seconds)), "shortWindowSeconds": min(5, seconds), "activeUploadRequests": gunshotActiveUploads.Load(), "uploadBodyBytesRead": gunshotUploadBytes.Load(), "windowSeconds": int64(seconds)}
}
