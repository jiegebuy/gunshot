const gunshotResumeThreshold int64 = 32 << 20
const gunshotResumeChunk int64 = 8 << 20
const gunshotUploadEndpoint = "https://photos.googleapis.com/data/upload/uploadmedia/interactive"

// Aim for twelve seconds of transfer per acknowledgement. Limit growth to
// avoid a fast sample causing an oversized request when the path slows down.
func gunshotNextChunk(current, sent int64, elapsed time.Duration) int64 {
	if elapsed <= 0 || sent <= 0 {
		return current
	}
	target := float64(sent) * float64(12*time.Second) / float64(elapsed)
	return max(1<<20, min(32<<20, current*2, int64(min(target, float64(32<<20)))))
}

// Session URLs are bearer capabilities. Keep this sidecar private beside the
// original; never include it in queue JSON, diagnostic exports or error text.
type gunshotUploadSession struct {
	Streaming   bool   `json:"streaming,omitempty"`
	Version     int    `json:"version"`
	Protocol    string `json:"protocol"`
	URL         string `json:"url"`
	Account     string `json:"account"`
	SHA1        string `json:"sha1"`
	Size        int64  `json:"size"`
	Offset      int64  `json:"offset"`
	Granularity int64  `json:"granularity"`
	ChunkTarget int64  `json:"chunkTarget,omitempty"`
	Finalize    []byte `json:"finalize,omitempty"`
	Updated     int64  `json:"updated"`
}

func gunshotSessionPath(path string) string {
	sum := sha256.Sum256([]byte(filepath.Base(path)))
	return filepath.Join(filepath.Dir(path), ".upload-"+hex.EncodeToString(sum[:8])+".json")
}

// Expose only a capability check to the queue, never the secret session URL.
func GunshotHasUploadCheckpoint(path string) bool {
	st, err := os.Stat(path)
	if err != nil {
		return false
	}
	raw, err := os.ReadFile(gunshotSessionPath(path))
	if err != nil || len(raw) > 1<<20 {
		return false
	}
	var s gunshotUploadSession
	return json.Unmarshal(raw, &s) == nil && s.Version == 1 && (s.Size == st.Size() || (s.Streaming && s.Size == -1 && s.Protocol == "scotty")) && s.Offset >= 0 && s.Offset <= st.Size() && gunshotSafeUploadURL(s.URL) && (s.Protocol == "scotty" || s.Protocol == "range")
}
func (a *Api) gunshotAccountKey() string {
	sum := sha256.Sum256([]byte(fmt.Sprintf("%s|%t|%t", a.authData, a.saver, a.useQuota)))
	return hex.EncodeToString(sum[:])
}
func gunshotSafeUploadURL(raw string) bool {
	u, err := url.Parse(raw)
	return err == nil && u.Scheme == "https" && u.User == nil && u.Fragment == "" && u.Port() == "" && u.Hostname() == "photos.googleapis.com" && strings.HasPrefix(u.Path, "/data/upload/")
}
func gunshotSaveSession(path string, s *gunshotUploadSession) error {
	s.Updated = time.Now().Unix()
	data, err := json.Marshal(s)
	if err != nil {
		return err
	}
	target := gunshotSessionPath(path)
	return gunshotAtomicPrivateFile(target, data)
}
func gunshotAtomicPrivateFile(target string, data []byte) error {
	f, err := os.CreateTemp(filepath.Dir(target), ".upload-write-")
	if err != nil {
		return err
	}
	defer os.Remove(f.Name())
	if err = f.Chmod(0600); err == nil {
		_, err = f.Write(data)
	}
	if err == nil {
		err = f.Sync()
	}
	closeErr := f.Close()
	if err == nil {
		err = closeErr
	}
	if err != nil {
		return err
	}
	if err = os.Rename(f.Name(), target); err != nil {
		return err
	}
	directory, err := os.Open(filepath.Dir(target))
	if err != nil {
		return err
	}
	defer directory.Close()
	return directory.Sync()
}
func (a *Api) gunshotLoadSession(path, hash string, size int64) (*gunshotUploadSession, error) {
	raw, err := os.ReadFile(gunshotSessionPath(path))
	if os.IsNotExist(err) {
		_, spoolErr := gunshotSpoolFloor(path, nil)
		return nil, spoolErr
	}
	if err != nil {
		return nil, err
	}
	var s gunshotUploadSession
	if len(raw) > 1<<20 {
		return nil, errors.New("invalid private upload checkpoint")
	}
	if json.Unmarshal(raw, &s) == nil && s.Streaming && s.Version == 1 && s.Size == -1 && s.Protocol == "scotty" && s.Account == a.gunshotAccountKey() && s.Offset >= 0 && s.Offset <= size && gunshotSafeUploadURL(s.URL) && s.Granularity > 0 && s.Granularity <= 64<<20 && len(s.Finalize) == 0 {
		if _, err := gunshotSpoolFloor(path, &s); err != nil {
			return nil, err
		}
		// The producer has sealed the full file. Adopt its earlier unknown-size
		// session; the final request supplies the now-verified complete hash.
		if hash == "" {
			return &s, nil
		}
		s.Streaming, s.Size, s.SHA1 = false, size, hash
		if err := gunshotSaveSession(path, &s); err != nil {
			return nil, err
		}
		return &s, nil
	}
	if len(raw) > 1<<20 || json.Unmarshal(raw, &s) != nil || s.Version != 1 || s.Size != size || s.Account != a.gunshotAccountKey() || (hash != "" && s.SHA1 != hash) || s.Offset < 0 || s.Offset > s.Size || !gunshotSafeUploadURL(s.URL) || (s.Protocol != "scotty" && s.Protocol != "range") {
		return nil, errors.New("invalid private upload checkpoint")
	}
	if len(s.Finalize) > 0 {
		if _, err = ParseScottyFinalizeToken(s.Finalize); err != nil {
			return nil, errors.New("invalid saved finalize response")
		}
	}
	if _, err := gunshotSpoolFloor(path, &s); err != nil {
		return nil, err
	}
	return &s, nil
}

// Negotiate resumption before sending file bytes. Existing small-file uploads
// retain their established path; large files require explicit server support.
func (a *Api) gunshotGetUploadTokenForFile(ctx context.Context, path, hash string, size int64) (string, error) {
	saved, err := a.gunshotLoadSession(path, hash, size)
	if err != nil {
		return "", err
	}
	if saved != nil {
		return saved.URL, nil
	}
	if size < gunshotResumeThreshold {
		return a.GetUploadToken(hash, size)
	}
	return a.gunshotStartResumable(ctx, path, hash, size)
}

func (a *Api) gunshotStartResumable(ctx context.Context, path, hash string, size int64) (string, error) {
	body, err := proto.Marshal(&generated.GetUploadToken{F1: 2, F2: 2, F3: 1, F4: 3, FileSizeBytes: max(0, size)})
	if err != nil {
		return "", err
	}
	c, cancel := context.WithTimeout(ctx, time.Minute)
	defer cancel()
	req, err := http.NewRequestWithContext(c, "POST", gunshotUploadEndpoint, bytes.NewReader(body))
	if err != nil {
		return "", err
	}
	bearer, err := a.BearerToken()
	if err != nil {
		return "", errors.New("upload session authorization unavailable")
	}
	req.Header.Set("Authorization", "Bearer "+bearer)
	req.Header.Set("User-Agent", a.userAgent)
	req.Header.Set("Accept-Language", a.language)
	req.Header.Set("Content-Type", "application/x-protobuf")
	if hash != "" {
		req.Header.Set("X-Goog-Hash", "sha1="+hash)
	}
	if size >= 0 {
		req.Header.Set("X-Upload-Content-Length", strconv.FormatInt(size, 10))
	}
	req.Header.Set("X-Goog-Upload-Protocol", "resumable")
	req.Header.Set("X-Goog-Upload-Command", "start")
	if size >= 0 {
		req.Header.Set("X-Goog-Upload-Header-Content-Length", strconv.FormatInt(size, 10))
	}
	req.Header.Set("X-Goog-Upload-Header-Content-Type", "application/octet-stream")
	client := *a.client
	client.CheckRedirect = func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }
	resp, err := client.Do(req)
	if err != nil {
		if ctx.Err() != nil {
			return "", ctx.Err()
		}
		return "", errors.New("upload session request interrupted")
	}
	defer resp.Body.Close()
	// Only status/header metadata is persisted in diagnostics; no session ID.
	gunshotRecordResumeProbe(path, resp.StatusCode, resp.Header)
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return "", fmt.Errorf("upload session negotiation returned HTTP %d", resp.StatusCode)
	}
	s := &gunshotUploadSession{Version: 1, Account: a.gunshotAccountKey(), SHA1: hash, Size: size, Streaming: size < 0, Granularity: 256 << 10}
	if raw := resp.Header.Get("X-Goog-Upload-URL"); raw != "" {
		s.Protocol = "scotty"
		s.URL = raw
		if g, e := strconv.ParseInt(resp.Header.Get("X-Goog-Upload-Chunk-Granularity"), 10, 64); e == nil {
			s.Granularity = g
		}
	} else if raw := resp.Header.Get("Location"); raw != "" {
		s.Protocol = "range"
		s.URL = raw
	} else {
		// Legacy Scotty may expose only its session ID. Probe that exact session
		// with an empty range query; never assume it supports offsets.
		token := resp.Header.Get("X-GUploader-UploadID")
		if token == "" {
			return "", errors.New("server did not offer a resumable upload session")
		}
		s.Protocol = "range"
		s.URL = gunshotUploadEndpoint + "?upload_id=" + url.QueryEscape(token)
	}
	if !gunshotSafeUploadURL(s.URL) || s.Granularity <= 0 || s.Granularity > 64<<20 {
		return "", errors.New("unsupported resumable session parameters")
	}
	if s.Streaming && s.Protocol != "scotty" {
		return "", errors.New("streaming requires a Scotty session")
	}
	if _, _, err = a.gunshotResumeRequest(ctx, path, s, 0, 0, nil); err != nil {
		return "", err
	}
	if err = gunshotSaveSession(path, s); err != nil {
		return "", err
	}
	return s.URL, nil
}
func gunshotRecordResumeProbe(path string, status int, h http.Header) {
	data, _ := json.Marshal(map[string]any{"status": status, "hasScottyURL": h.Get("X-Goog-Upload-URL") != "", "hasLocation": h.Get("Location") != "", "hasLegacyID": h.Get("X-GUploader-UploadID") != "", "uploadStatus": h.Get("X-Goog-Upload-Status"), "granularity": h.Get("X-Goog-Upload-Chunk-Granularity")})
	_ = os.WriteFile(filepath.Join(filepath.Dir(path), ".upload-protocol-diagnostic.json"), data, 0600)
}

// A query never creates a media item. Only server-acknowledged offsets advance
// the checkpoint; bytes read from a socket writer are not an acknowledgement.
func (a *Api) gunshotResumeRequest(ctx context.Context, path string, s *gunshotUploadSession, offset, n int64, reader io.Reader) (int64, []byte, error) {
	query := reader == nil
	method := "POST"
	if s.Protocol == "range" {
		method = "PUT"
	}
	c, cancel := context.WithTimeout(ctx, 90*time.Second)
	defer cancel()
	req, err := http.NewRequestWithContext(c, method, s.URL, reader)
	if err != nil {
		return 0, nil, errors.New("invalid upload request")
	}
	req.ContentLength = n
	token, err := a.BearerToken()
	if err != nil {
		return 0, nil, errors.New("upload authorization unavailable")
	}
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("User-Agent", a.userAgent)
	req.Header.Set("Content-Type", "application/octet-stream")
	if s.Protocol == "scotty" {
		command := "upload"
		if offset+n == s.Size {
			command = "upload, finalize"
			if s.SHA1 != "" {
				req.Header.Set("X-Goog-Hash", "sha1="+s.SHA1)
			}
		}
		if query {
			command = "query"
		}
		req.Header.Set("X-Goog-Upload-Command", command)
		if !query {
			req.Header.Set("X-Goog-Upload-Offset", strconv.FormatInt(offset, 10))
		}
	} else {
		value := fmt.Sprintf("bytes %d-%d/%d", offset, offset+n-1, s.Size)
		if query {
			value = fmt.Sprintf("bytes */%d", s.Size)
		}
		req.Header.Set("Content-Range", value)
	}
	client := *a.client
	client.CheckRedirect = func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }
	resp, err := client.Do(req)
	if err != nil {
		return 0, nil, errors.New("resumable transfer interrupted")
	}
	defer resp.Body.Close()
	if query || resp.StatusCode >= 400 {
		metadata, _ := json.Marshal(map[string]any{"protocol": s.Protocol, "query": query, "status": resp.StatusCode, "uploadStatus": resp.Header.Get("X-Goog-Upload-Status"), "received": resp.Header.Get("X-Goog-Upload-Size-Received"), "range": resp.Header.Get("Range")})
		_ = os.WriteFile(filepath.Join(filepath.Dir(path), ".upload-transfer-diagnostic.json"), metadata, 0600)
	}
	if resp.StatusCode == 404 || resp.StatusCode == 410 {
		if spool, e := gunshotLoadSpool(path); e != nil || spool != nil {
			return 0, nil, ErrGunshotStreamLost
		}
		return 0, nil, errors.New("upload session expired; manual restart required")
	}
	if resp.StatusCode != 308 && (resp.StatusCode < 200 || resp.StatusCode >= 300) {
		return 0, nil, fmt.Errorf("resumable transfer returned HTTP %d", resp.StatusCode)
	}
	if location := resp.Header.Get("Location"); location != "" {
		if !gunshotSafeUploadURL(location) {
			return 0, nil, errors.New("invalid session redirect")
		}
		s.URL = location
	}
	raw, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
	if err != nil {
		return 0, nil, errors.New("upload response interrupted")
	}
	if len(raw) > 0 {
		if final, e := ParseScottyFinalizeToken(raw); e == nil {
			return s.Size, final.Raw, nil
		}
	}
	accepted := int64(-1)
	if s.Protocol == "scotty" {
		if resp.Header.Get("X-Goog-Upload-Status") != "active" {
			return 0, nil, errors.New("upload finalized without recoverable receipt")
		}
		if value := resp.Header.Get("X-Goog-Upload-Size-Received"); value != "" {
			accepted, err = strconv.ParseInt(value, 10, 64)
			if err != nil {
				return 0, nil, errors.New("invalid server offset")
			}
		} else if query {
			return 0, nil, errors.New("server omitted resumable offset")
		} else if resp.StatusCode == http.StatusOK && (s.Streaming || offset+n < s.Size) {
			// A successful ordinary upload command acknowledges this entire
			// chunk. Query only after interruption or restart, not every chunk.
			accepted = offset + n
		}
	} else {
		if resp.StatusCode != 308 {
			return 0, nil, errors.New("server does not support range resumption")
		}
		value := resp.Header.Get("Range")
		if value == "" {
			accepted = 0
		} else if strings.HasPrefix(value, "bytes=0-") {
			end, e := strconv.ParseInt(strings.TrimPrefix(value, "bytes=0-"), 10, 64)
			if e != nil || end < 0 || end >= s.Size {
				return 0, nil, errors.New("invalid acknowledged range")
			}
			accepted = end + 1
		} else {
			return 0, nil, errors.New("invalid acknowledged range")
		}
	}
	if (!s.Streaming && accepted > s.Size) || accepted < -1 {
		return 0, nil, errors.New("invalid server offset")
	}
	return accepted, nil, nil
}

func (a *Api) UploadFileWithProgress(ctx context.Context, path, uploadToken string, onProgress UploadProgressCallback) (ScottyFinalizeToken, error) {
	st, err := os.Stat(path)
	if err != nil {
		return ScottyFinalizeToken{}, err
	}
	spool, err := gunshotLoadSpool(path)
	if err != nil {
		return ScottyFinalizeToken{}, err
	}
	if spool != nil && !GunshotHasUploadCheckpoint(path) {
		return ScottyFinalizeToken{}, ErrGunshotStreamLost
	}
	if st.Size() < gunshotResumeThreshold && !GunshotHasUploadCheckpoint(path) {
		return a.gunshotUploadFileLegacy(ctx, path, uploadToken, onProgress)
	}
	s, err := a.gunshotLoadSession(path, "", st.Size())
	if err != nil {
		return ScottyFinalizeToken{}, err
	}
	if s != nil && s.Streaming {
		digest, e := CalculateSHA1(ctx, path)
		if e != nil {
			return ScottyFinalizeToken{}, e
		}
		s, err = a.gunshotLoadSession(path, base64.StdEncoding.EncodeToString(digest), st.Size())
		if err != nil {
			return ScottyFinalizeToken{}, err
		}
	}
	if s == nil {
		hash, e := CalculateSHA1(ctx, path)
		if e != nil {
			return ScottyFinalizeToken{}, e
		}
		if _, e = a.gunshotGetUploadTokenForFile(ctx, path, base64.StdEncoding.EncodeToString(hash), st.Size()); e != nil {
			return ScottyFinalizeToken{}, e
		}
		s, err = a.gunshotLoadSession(path, "", st.Size())
		if err != nil || s == nil {
			return ScottyFinalizeToken{}, errors.New("large upload has no resumable checkpoint")
		}
	}
	if len(s.Finalize) > 0 {
		return ParseScottyFinalizeToken(s.Finalize)
	}
	f, err := os.Open(path)
	if err != nil {
		return ScottyFinalizeToken{}, err
	}
	defer f.Close()
	needQuery := true
	chunkTarget := gunshotResumeChunk
	failures := 0
	attempt := 1
	for {
		if err = ctx.Err(); err != nil {
			return ScottyFinalizeToken{}, err
		}
		var accepted int64
		var final []byte
		started := time.Now()
		sentChunk := int64(0)
		if needQuery {
			accepted, final, err = a.gunshotResumeRequest(ctx, path, s, 0, 0, nil)
		} else {
			if s.Offset >= s.Size {
				return ScottyFinalizeToken{}, errors.New("awaiting upload receipt; no bytes retransmitted")
			}
			chunk := max(chunkTarget, s.Granularity)
			chunk = (chunk / s.Granularity) * s.Granularity
			n := min(chunk, s.Size-s.Offset)
			sentChunk = n
			start := s.Offset
			currentAttempt := attempt
			reader := NewProgressReader(io.NewSectionReader(f, start, n), n, func(sent, total int64) {
				if onProgress != nil {
					onProgress(start+sent, s.Size, currentAttempt)
				}
			})
			accepted, final, err = a.gunshotResumeRequest(ctx, path, s, start, n, reader)
		}
		if err != nil {
			if ctx.Err() != nil {
				return ScottyFinalizeToken{}, ctx.Err()
			}
			failures++
			attempt++
			chunkTarget = max(1<<20, chunkTarget/2)
			if failures > 3 {
				return ScottyFinalizeToken{}, err
			}
			needQuery = true
			select {
			case <-ctx.Done():
				return ScottyFinalizeToken{}, ctx.Err()
			case <-time.After(time.Duration(failures) * time.Second):
			}
			continue
		}
		if len(final) > 0 {
			s.Offset = s.Size
			s.Finalize = final
			if err = gunshotSaveSession(path, s); err != nil {
				return ScottyFinalizeToken{}, err
			}
			if onProgress != nil {
				onProgress(s.Size, s.Size, attempt)
			}
			return ParseScottyFinalizeToken(final)
		}
		if accepted >= 0 {
			if floor, e := gunshotSpoolFloor(path, s); e != nil || accepted < floor {
				return ScottyFinalizeToken{}, ErrGunshotStreamLost
			}
			if !needQuery && accepted > s.Offset && sentChunk > 0 {
				chunkTarget = gunshotNextChunk(chunkTarget, min(sentChunk, accepted-s.Offset), time.Since(started))
			}
			if !needQuery && accepted <= s.Offset {
				failures++
				if failures > 3 {
					return ScottyFinalizeToken{}, errors.New("server offset did not advance")
				}
			} else if accepted > s.Offset {
				failures = 0
			}
			s.Offset = accepted
			if err = gunshotSaveSession(path, s); err != nil {
				return ScottyFinalizeToken{}, err
			}
			if onProgress != nil {
				onProgress(s.Offset, s.Size, attempt)
			}
			needQuery = false
		} else {
			failures++
			if failures > 3 {
				return ScottyFinalizeToken{}, errors.New("server did not acknowledge chunk progress")
			}
			needQuery = true
		}
	}
}
