func TestGunshotTransportMetricsOmitSessionAndAccount(t *testing.T) {
	request, _ := http.NewRequest("POST", "https://photos.googleapis.com/data/upload/uploadmedia/interactive?upload_id=private-session-token", strings.NewReader("private-file-bytes"))
	request.Header.Set("Authorization", "Bearer private-access-token")
	transport := contextTransport{context.Background(), gunshotRoundTripper(func(r *http.Request) (*http.Response, error) {
		io.Copy(io.Discard, r.Body)
		return &http.Response{StatusCode: 200, ProtoMajor: 2, Body: io.NopCloser(strings.NewReader("ok"))}, nil
	})}
	reply, err := transport.RoundTrip(request)
	if err != nil {
		t.Fatal(err)
	}
	reply.Body.Close()
	snapshot := GunshotTransportSnapshot()
	raw, _ := json.Marshal(snapshot)
	for _, private := range []string{"private-session-token", "private-access-token", "private-file-bytes", "upload_id"} {
		if strings.Contains(string(raw), private) {
			t.Fatal("sensitive data in timing diagnostic")
		}
	}
	entries := snapshot["requests"].(map[string]any)
	entry := entries["uploadBody"].(map[string]any)
	if entry["requests"].(int64) < 1 || entry["requestBodyBytes"].(int64) < int64(len("private-file-bytes")) {
		t.Fatal("missing request metrics")
	}
	if entry["outcomes"].(map[string]int64)["200"] < 1 {
		t.Fatal("missing request outcome")
	}
}

func TestGunshotRequestOutcomeClasses(t *testing.T) {
	for want, err := range map[string]error{"canceled": context.Canceled, "timeout": context.DeadlineExceeded, "eof": io.ErrUnexpectedEOF, "error": errors.New("tls")} {
		if got := gunshotRequestOutcome(nil, err); got != want {
			t.Fatalf("%v classified as %s, want %s", err, got, want)
		}
	}
	if got := gunshotRequestOutcome(&http.Response{StatusCode: 503}, nil); got != "503" {
		t.Fatal(got)
	}
}

func TestGunshotJobsShareConnectionsWithoutSharingClients(t *testing.T) {
	var connections atomic.Int32
	server := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { _, _ = w.Write([]byte("ok")) }))
	server.Config.ConnState = func(_ net.Conn, state http.ConnState) {
		if state == http.StateNew {
			connections.Add(1)
		}
	}
	server.Start()
	defer server.Close()
	first, err := NewHTTPClientWithProxy("")
	if err != nil {
		t.Fatal(err)
	}
	second, err := NewHTTPClientWithProxy("")
	if err != nil {
		t.Fatal(err)
	}
	if first == second || first.Transport != second.Transport {
		t.Fatal("clients/accounts must be separate but transport reusable")
	}
	for _, client := range []*http.Client{first, second} {
		response, err := client.Get(server.URL)
		if err != nil {
			t.Fatal(err)
		}
		io.Copy(io.Discard, response.Body)
		response.Body.Close()
	}
	if connections.Load() != 1 {
		t.Fatal("each job opened a new connection", connections.Load())
	}
}

type gunshotRoundTripper func(*http.Request) (*http.Response, error)

func TestGunshotConcurrentUploadsUseSeparateConnections(t *testing.T) {
	var connections atomic.Int32
	arrived := make(chan int, 2)
	release := make(chan struct{})
	server := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		arrived <- r.ProtoMajor
		select {
		case <-release:
		case <-r.Context().Done():
		}
		_, _ = w.Write([]byte("ok"))
	}))
	server.EnableHTTP2 = true // The server would accept h2; the client must not multiplex.
	server.Config.ConnState = func(_ net.Conn, state http.ConnState) {
		if state == http.StateNew {
			connections.Add(1)
		}
	}
	server.StartTLS()
	defer server.Close()
	unblock := sync.OnceFunc(func() { close(release) })
	defer unblock() // Runs before Close, so a failed assertion cannot hang the server.
	transport := gunshotSharedTransport("separate-connections-test", server.Client().Transport.(*http.Transport).Clone())
	client := &http.Client{Transport: transport}
	done := make(chan error, 2)
	for range 2 {
		go func() {
			response, err := client.Post(server.URL, "application/octet-stream", strings.NewReader("chunk"))
			if err == nil {
				io.Copy(io.Discard, response.Body)
				response.Body.Close()
			}
			done <- err
		}()
	}
	for range 2 {
		select {
		case proto := <-arrived:
			if proto != 1 {
				t.Fatal("upload negotiated HTTP/2", proto)
			}
		case <-time.After(5 * time.Second):
			t.Fatal("concurrent uploads were serialized on one connection")
		}
	}
	unblock()
	for range 2 {
		if err := <-done; err != nil {
			t.Fatal(err)
		}
	}
	if connections.Load() != 2 {
		t.Fatal("concurrent uploads shared a connection", connections.Load())
	}
}

func (f gunshotRoundTripper) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

type gunshotContextReader struct {
	ctx context.Context
	io.Reader
}

func (b gunshotContextReader) Read(p []byte) (int, error) {
	if err := b.ctx.Err(); err != nil {
		return 0, err
	}
	return b.Reader.Read(p)
}
func (b gunshotContextReader) Close() error { return nil }

func gunshotAwaitCancellation(t *testing.T, ctx context.Context) {
	t.Helper()
	select {
	case <-ctx.Done():
	case <-time.After(time.Second):
		t.Fatal("transport lost cancellation")
	}
}
func TestGunshotCommitWaitStopsWithoutRetry(t *testing.T) {
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	api := &Api{client: &http.Client{Transport: contextTransport{ctx: ctx, base: http.DefaultTransport}}}
	started := time.Now()
	if !errors.Is(gunshotCommitWait(api, time.Hour), context.Canceled) {
		t.Fatal("ignored cancellation")
	}
	if time.Since(started) > time.Second {
		t.Fatal("waited through backoff")
	}
	if _, err := api.commitSerialized(nil); !errors.Is(err, context.Canceled) {
		t.Fatal("retried cancelled commit", err)
	}
}
func TestGunshotTransportPreservesRequestContext(t *testing.T) {
	type key struct{}
	parent, cancel := context.WithTimeout(context.WithValue(context.Background(), key{}, "request value"), time.Minute)
	defer cancel()
	request, err := http.NewRequestWithContext(parent, http.MethodGet, "https://example.invalid", nil)
	if err != nil {
		t.Fatal(err)
	}
	var transported context.Context
	transport := contextTransport{context.Background(), gunshotRoundTripper(func(r *http.Request) (*http.Response, error) {
		transported = r.Context()
		if transported.Value(key{}) != "request value" {
			t.Error("request values lost")
		}
		want, _ := parent.Deadline()
		got, ok := transported.Deadline()
		if !ok || !got.Equal(want) {
			t.Error("request/client deadline lost")
		}
		return &http.Response{Body: gunshotContextReader{transported, strings.NewReader("body")}}, nil
	})}
	response, err := transport.RoundTrip(request)
	if err != nil {
		t.Fatal(err)
	}
	defer response.Body.Close()
	if request.Context() != parent {
		t.Fatal("original request was mutated")
	}
	if transported.Err() != nil {
		t.Fatal("cancelled before reading response body")
	}
	cancel()
	gunshotAwaitCancellation(t, transported)
	if _, err := io.ReadAll(response.Body); !errors.Is(err, context.Canceled) {
		t.Fatalf("body ignored request cancellation: %v", err)
	}
}
func TestGunshotTransportBodyLifetime(t *testing.T) {
	for _, end := range []string{"upload cancellation", "close", "eof", "transport error", "no body", "already cancelled"} {
		t.Run(end, func(t *testing.T) {
			upload, cancel := context.WithCancel(context.Background())
			defer cancel()
			if end == "already cancelled" {
				cancel()
			}
			var transported context.Context
			transport := contextTransport{upload, gunshotRoundTripper(func(r *http.Request) (*http.Response, error) {
				transported = r.Context()
				if end == "transport error" {
					return nil, io.ErrUnexpectedEOF
				}
				if end == "no body" {
					return &http.Response{}, nil
				}
				return &http.Response{Body: gunshotContextReader{transported, strings.NewReader("body")}}, nil
			})}
			request, _ := http.NewRequest(http.MethodGet, "https://example.invalid", nil)
			response, err := transport.RoundTrip(request)
			if end == "already cancelled" {
				if !errors.Is(err, context.Canceled) || response != nil {
					t.Fatalf("sent cancelled upload: %v", err)
				}
				return
			}
			if end == "transport error" {
				if !errors.Is(err, io.ErrUnexpectedEOF) {
					t.Fatal(err)
				}
			} else if err != nil {
				t.Fatal(err)
			}
			if response != nil && response.Body != nil {
				defer response.Body.Close()
			}
			switch end {
			case "upload cancellation":
				cancel()
			case "close":
				if transported.Err() != nil {
					t.Fatal("body cancelled at headers")
				}
				if err := response.Body.Close(); err != nil {
					t.Fatal(err)
				}
			case "eof":
				data, err := io.ReadAll(response.Body)
				if err != nil || string(data) != "body" {
					t.Fatalf("response body cut short: %q %v", data, err)
				}
			}
			gunshotAwaitCancellation(t, transported)
		})
	}
}
