func TestGrowingFileUploadsBeforeEOFAndResumesInNewClient(t *testing.T) {
	a, server, path := resumeFixture(t)
	data := bytes.Repeat([]byte("original-media-byte-pattern"), 95001)
	server.size = int64(len(data))
	prefix := int64(768 << 10)
	if err := os.WriteFile(path, data[:prefix], 0600); err != nil {
		t.Fatal(err)
	}
	a.client.Transport = gunshotRoundTripper(func(r *http.Request) (*http.Response, error) {
		if r.Header.Get("X-Goog-Upload-Command") == "start" && (r.Header.Get("X-Goog-Hash") != "" || r.Header.Get("X-Goog-Upload-Header-Content-Length") != "") {
			t.Fatal("required full-file metadata before EOF")
		}
		if r.Header.Get("X-Goog-Upload-Command") == "upload, finalize" {
			hash := sha1.Sum(data)
			if r.Header.Get("X-Goog-Hash") != "sha1="+base64.StdEncoding.EncodeToString(hash[:]) {
				t.Fatal("missing final hash")
			}
		}
		if r.Header.Get("X-Goog-Upload-Command") == "upload" || r.Header.Get("X-Goog-Upload-Command") == "upload, finalize" {
			b, err := io.ReadAll(r.Body)
			if err != nil {
				t.Fatal(err)
			}
			if !bytes.Equal(b, data[server.received:server.received+int64(len(b))]) {
				t.Fatal("wrong bytes")
			}
			r.Body = io.NopCloser(bytes.NewReader(b))
		}
		return server.RoundTrip(r)
	})
	ack, err := a.gunshotPreupload(context.Background(), path, prefix)
	if err != nil || ack <= 0 || ack >= prefix || server.received != ack {
		t.Fatalf("no verified pre-EOF upload: %d %v", ack, err)
	}
	st, _ := os.Stat(path)
	if st.Size() != prefix || !GunshotHasUploadCheckpoint(path) {
		t.Fatal("expected incomplete retained file and session")
	}
	// Finish producing, then create a new API object as after process restart.
	f, err := os.OpenFile(path, os.O_WRONLY|os.O_APPEND, 0600)
	if err != nil {
		t.Fatal(err)
	}
	_, err = f.Write(data[prefix:])
	f.Close()
	if err != nil {
		t.Fatal(err)
	}
	next := &Api{client: a.client, authData: a.authData, authResponseCache: a.authResponseCache}
	ack2, err := next.gunshotPreupload(context.Background(), path, int64(len(data)))
	if err != nil || ack2 <= ack {
		t.Fatal("resume failed", ack2, err)
	}
	if server.starts != 1 {
		t.Fatal("replaced existing upload session")
	}
	hash := sha1.Sum(data)
	token, err := next.gunshotGetUploadTokenForFile(context.Background(), path, base64.StdEncoding.EncodeToString(hash[:]), int64(len(data)))
	if err != nil {
		t.Fatal(err)
	}
	final, err := next.UploadFileWithProgress(context.Background(), path, token, nil)
	if err != nil || len(final.Raw) == 0 || server.received != int64(len(data)) || server.starts != 1 {
		t.Fatal("sealed stream did not finish in same session", err)
	}
}

func TestPreuploadRejectsServerAheadOfProducer(t *testing.T) {
	a, s, path := resumeFixture(t)
	if err := os.Truncate(path, 1<<20); err != nil {
		t.Fatal(err)
	}
	s.received = 2 << 20
	if _, err := a.gunshotPreupload(context.Background(), path, 1<<20); err == nil {
		t.Fatal("accepted bytes outside verified prefix")
	}
	if s.chunks != 0 {
		t.Fatal("sent data despite invalid acknowledgement")
	}
}

func TestPreuploadPersistsAdaptiveTargetAndRecoversPartialFailure(t *testing.T) {
	synctest.Test(t, func(t *testing.T) {
		a, server, path := resumeFixture(t)
		var lengths []int64
		a.client.Transport = gunshotRoundTripper(func(r *http.Request) (*http.Response, error) {
			if r.Header.Get("X-Goog-Upload-Command") == "upload" {
				lengths = append(lengths, r.ContentLength)
				switch len(lengths) {
				case 1:
					time.Sleep(48 * time.Second)
				case 3:
					// The server accepted a prefix before the response was lost.
					n, err := io.CopyN(io.Discard, r.Body, 256<<10)
					if err != nil {
						t.Fatal(err)
					}
					server.received += n
					return nil, io.ErrUnexpectedEOF
				default:
					time.Sleep(time.Second)
				}
			}
			return server.RoundTrip(r)
		})
		var ack int64
		for attempt, expected := range []int64{2 << 20, 4 << 20, 2 << 20, 4 << 20} {
			// Each worker/restart has a new API; the learned target must persist.
			next := &Api{client: a.client, authData: a.authData, authResponseCache: a.authResponseCache}
			previous := ack
			var err error
			ack, err = next.gunshotPreupload(context.Background(), path, server.size)
			if (attempt == 2) != (err != nil) {
				t.Fatal("unexpected transfer result", attempt, err)
			}
			s, loadErr := next.gunshotLoadSession(path, "", server.size)
			if loadErr != nil || s.ChunkTarget != expected {
				t.Fatal("adaptive target not persisted", attempt, loadErr, s)
			}
			if attempt == 2 && (ack != previous || s.Offset != previous || server.received <= ack) {
				t.Fatal("guessed acknowledgement after interrupted request")
			}
		}
		if len(lengths) != 4 || lengths[0] != 8<<20 || lengths[1] != 2<<20 || lengths[2] != 4<<20 || lengths[3] != 2<<20 || server.starts != 1 {
			t.Fatal("wrong chunk targets or replaced resumable session", lengths, server.starts)
		}
	})
}
